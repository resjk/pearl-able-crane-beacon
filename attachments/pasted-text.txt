def load_images(cell_size):
    path = os.path.join(BASE_DIR, "assets", "pieces")
    if not os.path.exists(path): return
    for filename in os.listdir(path):
        if filename.endswith(".png"):
            piece_name = filename[:-4]
            try:
                img = pygame.image.load(os.path.join(path, filename))
                img = img.convert_alpha()
                bg_color = img.get_at((0, 0))
                img.set_colorkey(bg_color)
                img = pygame.transform.scale(img, (cell_size, cell_size))
                IMAGES[piece_name] = img
            except Exception:
                pass

def get_cjk_font(size, bold=False):
    candidates = ["notosanscjksc", "notosanscjk", "notosansmonocjksc", "wenquanyizenhei", "wenquanyimicrohei", "microsoftyahei", "simsun", "simhei", "arialunicodems"]
    for name in candidates:
        path = pygame.font.match_font(name, bold=bold)
        if path: return pygame.font.Font(path, size)
    return pygame.font.SysFont("Arial", size, bold=bold)

def make_start_board():
    board = [[None]*8 for _ in range(8)]
    back_rank = ["R","N","B","Q","K","B","N","R"]
    for c in range(8):
        board[0][c] = {"type": "b" + back_rank[c], "armor": 0, "projectile": False, "gold_projectile": False, "moved": False, "frozen": 0}
        board[1][c] = {"type": "bP", "armor": 0, "projectile": False, "gold_projectile": False, "moved": False, "frozen": 0}
        board[6][c] = {"type": "wP", "armor": 0, "projectile": False, "gold_projectile": False, "moved": False, "frozen": 0}
        board[7][c] = {"type": "w" + back_rank[c], "armor": 0, "projectile": False, "gold_projectile": False, "moved": False, "frozen": 0}
    return board

def copy_board(board):
    return [[cell.copy() if cell else None for cell in row] for row in board]

def find_king(board, color):
    for r in range(8):
        for c in range(8):
            if board[r][c] and board[r][c]["type"] == color + "K":
                return r, c
    return None

def is_valid_move(board, from_r, from_c, to_r, to_c, last_move=None, buffs=None, teleports=None):
    piece = board[from_r][from_c]
    if piece is None: return False
    color = piece["type"][0]
    piece_type = piece["type"][1:]
    target = board[to_r][to_c]
    
    if target is not None and target["type"][0] == color:
        if not (piece.get("gold_projectile") and target is not None):
            return False
    if from_r == to_r and from_c == to_c: return False
    if teleports and (to_r, to_c) in teleports: return False
    
    dr = to_r - from_r
    dc = to_c - from_c
    
    if piece.get("projectile", False) or piece.get("gold_projectile", False):
        if dr != 0 and dc != 0 and abs(dr) != abs(dc): return False
        step_r = (dr > 0) - (dr < 0); step_c = (dc > 0) - (dc < 0)
        curr_r, curr_c = from_r + step_r, from_c + step_c
        while (curr_r, curr_c) != (to_r, to_c):
            if board[curr_r][curr_c] is not None: return False
            curr_r += step_r; curr_c += step_c
        return True

    if piece_type == "K":
        if abs(dr) <= 1 and abs(dc) <= 1:
            if target is not None and target.get("armor", 0) > 0: return False
            return True
        if dc == 2 and dr == 0 and not piece["moved"]:
            rook = board[from_r][7]
            if rook and rook["type"] == color + "R" and not rook["moved"]:
                if all(board[from_r][c] is None for c in range(from_c + 1, 7)):
                    if not is_king_in_check(board, color, buffs):
                        temp_board = copy_board(board); temp_board[from_r][from_c + 1] = temp_board[from_r][from_c]; temp_board[from_r][from_c] = None
                        if not is_king_in_check(temp_board, color, buffs): return True
        if dc == -2 and dr == 0 and not piece["moved"]:
            rook = board[from_r][0]
            if rook and rook["type"] == color + "R" and not rook["moved"]:
                if all(board[from_r][c] is None for c in range(1, from_c)):
                    if not is_king_in_check(board, color, buffs):
                        temp_board = copy_board(board); temp_board[from_r][from_c - 1] = temp_board[from_r][from_c]; temp_board[from_r][from_c] = None
                        if not is_king_in_check(temp_board, color, buffs): return True
        return False

    if piece_type == "R":
        if dr != 0 and dc != 0: return False
        step_r = (dr > 0) - (dr < 0); step_c = (dc > 0) - (dc < 0)
        curr_r, curr_c = from_r + step_r, from_c + step_c
        while (curr_r, curr_c) != (to_r, to_c):
            if board[curr_r][curr_c] is not None: return False
            curr_r += step_r; curr_c += step_c
        return True
        
    if piece_type == "B":
        if abs(dr) != abs(dc): return False
        step_r = (dr > 0) - (dr < 0); step_c = (dc > 0) - (dc < 0)
        curr_r, curr_c = from_r + step_r, from_c + step_c
        while (curr_r, curr_c) != (to_r, to_c):
            if board[curr_r][curr_c] is not None: return False
            curr_r += step_r; curr_c += step_c
        return True

    if piece_type == "Q":
        if dr != 0 and dc != 0 and abs(dr) != abs(dc): return False
        step_r = (dr > 0) - (dr < 0); step_c = (dc > 0) - (dc < 0)
        curr_r, curr_c = from_r + step_r, from_c + step_c
        while (curr_r, curr_c) != (to_r, to_c):
            if board[curr_r][curr_c] is not None: return False
            curr_r += step_r; curr_c += step_c
        return True

    if piece_type == "N":
        return (abs(dr) == 2 and abs(dc) == 1) or (abs(dr) == 1 and abs(dc) == 2)

    if piece_type == "P":
        direction = -1 if color == "w" else 1
        start_row = 6 if color == "w" else 1
        can_move_2 = buffs and "pawn_rush" in buffs.get(color, [])
        cruel = buffs and "cruel_pawns" in buffs.get(color, [])
        peaceful = buffs and "peaceful_pawns" in buffs.get(color, [])
        
        if dc == 0 and dr == direction: return target is None
        if dc == 0 and dr == 2 * direction and (from_r == start_row or can_move_2):
            mid_r = from_r + direction
            return board[mid_r][from_c] is None and target is None
        if abs(dc) == 1 and dr == direction:
            if peaceful: return False
            if target is not None and target["type"][0] != color: return True
            if target is None and last_move:
                if last_move["from_r"] == from_r + direction * -2 and last_move["from_c"] == to_c:
                    if board[from_r][to_c] and board[from_r][to_c]["type"][1] == "P": return True
        if cruel:
            if dc in [-1, 0, 1] and dr == direction:
                return target is None or target["type"][0] != color
    return False

def is_king_in_check(board, color, buffs=None):
    king_pos = find_king(board, color)
    if not king_pos: return False
    enemy_color = "b" if color == "w" else "w"
    for r in range(8):
        for c in range(8):
            piece = board[r][c]
            if piece and piece["type"][0] == enemy_color and piece.get("frozen", 0) == 0:
                if is_valid_move(board, r, c, king_pos[0], king_pos[1], None, buffs):
                    return True
    return False

def is_checkmate_or_stalemate(board, color, last_move, buffs):
    for r in range(8):
        for c in range(8):
            piece = board[r][c]
            if piece and piece["type"][0] == color and piece.get("frozen", 0) == 0:
                for tr in range(8):
                    for tc in range(8):
                        if is_valid_move(board, r, c, tr, tc, last_move, buffs):
                            temp_board = copy_board(board)
                            temp_board[tr][tc] = temp_board[r][c]
                            temp_board[r][c] = None
                            if not is_king_in_check(temp_board, color, buffs):
                                return False
    if is_king_in_check(board, color, buffs): return "checkmate"
    else: return "stalemate"

def draw_button(screen, rect, text, font, mouse_pos, disabled=False):
    if disabled: pygame.draw.rect(screen, BTN_DISABLED, rect, border_radius=10)
    else:
        color = BTN_HOVER if rect.collidepoint(mouse_pos) else BTN_COLOR
        pygame.draw.rect(screen, color, rect, border_radius=10)
    text_color = (100, 100, 100) if disabled else TEXT_COLOR
    text_surf = font.render(text, True, text_color)
    screen.blit(text_surf, text_surf.get_rect(center=rect.center))

def draw_game(screen, state_data, font_title, font_info, font_coords, font_cjk_small):

    turn = state_data["turn"]; check_status = state_data["check_status"]; check_alpha = state_data["check_alpha"]
    items = state_data["items"]; is_book_open = state_data["is_book_open"]
    proj_anim = state_data.get("proj_anim", None); bomb_anim = state_data.get("bomb_anim", None)
    buffs = state_data["buffs"]; blind_turns = state_data.get("blind_turns", 0)
    frozen_zones = state_data["frozen_zones"]; teleports = state_data["teleports"]
    trip_mode = state_data.get("trip_mode", False); targeting_spell = state_data.get("targeting_spell", None)
    used_spells = state_data.get("used_spells", set()); move_counter = state_data.get("move_counter", 1)
    last_spell_info = state_data.get("last_spell_info", "")
    mouse_pos = pygame.mouse.get_pos()
    
    if trip_mode: screen.fill((random.randint(0,255), random.randint(0,255), random.randint(0,255)))
    else: screen.fill(BG_COLOR)
    
    # 1. РАДИУС СПОСОБНОСТЕЙ
    hover_r, hover_c = None, None
    if targeting_spell and (BOARD_X <= mouse_pos[0] < BOARD_X + BOARD_W and BOARD_Y <= mouse_pos[1] < BOARD_Y + BOARD_H):
        hover_c = (mouse_pos[0] - BOARD_X) // CELL_SIZE
        hover_r = (mouse_pos[1] - BOARD_Y) // CELL_SIZE
    
    if targeting_spell and hover_r is not None:
        s_type = targeting_spell[1]
        s_overlay = pygame.Surface((CELL_SIZE, CELL_SIZE), pygame.SRCALPHA)
        s_overlay.fill((255, 50, 50, 120))
        if s_type in ["bombard_row", "bombard_col"]:
            for c in range(8): screen.blit(s_overlay, (BOARD_X + c*CELL_SIZE, BOARD_Y + hover_r*CELL_SIZE))
            for r in range(8): screen.blit(s_overlay, (BOARD_X + hover_c*CELL_SIZE, BOARD_Y + r*CELL_SIZE))
        elif s_type in ["freeze", "bomb"]:
            for fr in range(hover_r - 1, hover_r + 2):
                for fc in range(hover_c - 1, hover_c + 2):
                    if 0 <= fr < 8 and 0 <= fc < 8:
                        screen.blit(s_overlay, (BOARD_X + fc*CELL_SIZE, BOARD_Y + fr*CELL_SIZE))

        for c in range(BOARD_SIZE):

    # 10.5 АНИМАЦИЯ БОМБАРДИРОВКИ

def apply_spell(spell_type, board, turn, buffs, state_data, target1=None, target2=None):
    color = turn
    
    if spell_type == "teleport":
        state_data["teleports"] = [target1, target2]
        
    elif spell_type == "give_projectile":
        r, c = target1
        if board[r][c]: board[r][c]["projectile"] = True
        
    elif spell_type == "kill_any":
        r, c = target1
        if board[r][c] and board[r][c]["type"][1] != "K":
            board[r][c] = None
        
    elif spell_type == "swap":
        r1, c1 = target1; r2, c2 = target2
        if board[r1][c1] and board[r2][c2]:
            if board[r1][c1]["type"][1] != "K" and board[r2][c2]["type"][1] != "K":
                temp = copy_board(board)
                temp[r1][c1], temp[r2][c2] = temp[r2][c2], temp[r1][c1]
                if not is_king_in_check(temp, color, buffs):
                    board[r1][c1], board[r2][c2] = board[r2][c2], board[r1][c1]
        
    elif spell_type == "rewind":
        if len(state_data["history"]) >= 3:
            state_data["board"] = state_data["history"][-3]["board"]
            state_data["turn"] = state_data["history"][-3]["turn"]
            state_data["move_counter"] = max(1, state_data["move_counter"] - 3)
            state_data["history"] = state_data["history"][:-3]
        
    elif spell_type == "pawn_rush":
        buffs[color].append("pawn_rush")
        
    elif spell_type == "cruel_pawns":
        buffs[color].append("cruel_pawns")
        
    elif spell_type == "peaceful_pawns":
        buffs[color].append("peaceful_pawns")
        
    elif spell_type == "bombard_row":
        r = target1[0]
        can_attack = True
        for c in range(8):
            if board[r][c] and board[r][c]["type"][1] == "K":
                can_attack = False
                break
        if can_attack:
            king_pos = find_king(board, color)
            if king_pos:
                state_data["bomb_anim"] = {"from_r": king_pos[0], "from_c": king_pos[1], "to_r": r, "to_c": target1[1], "progress": 0.0}
            for c in range(8):
                if board[r][c]: board[r][c] = None
        
    elif spell_type == "bombard_col":
        c = target1[1]
        can_attack = True
        for r in range(8):
            if board[r][c] and board[r][c]["type"][1] == "K":
                can_attack = False
                break
        if can_attack:
            king_pos = find_king(board, color)
            if king_pos:
                state_data["bomb_anim"] = {"from_r": king_pos[0], "from_c": king_pos[1], "to_r": target1[0], "to_c": c, "progress": 0.0}
            for r in range(8):
                if board[r][c]: board[r][c] = None
        
    elif spell_type == "freeze":
        r, c = target1
        r_start = max(0, r - 1)
        c_start = max(0, c - 1)
        state_data["frozen_zones"].append({"r": r_start, "c": c_start, "turns": 2})
        for fr in range(r_start, r_start+3):
            for fc in range(c_start, c_start+3):
                if 0 <= fr < 8 and 0 <= fc < 8 and board[fr][fc]:
                    board[fr][fc]["frozen"] = 2
        
    elif spell_type == "bomb":
        r, c = target1
        r_start = max(0, r - 1)
        c_start = max(0, c - 1)
        can_attack = True
        for fr in range(r_start, r_start+3):
            for fc in range(c_start, c_start+3):
                if 0 <= fr < 8 and 0 <= fc < 8 and board[fr][fc]:
                    if board[fr][fc]["type"][1] == "K":
                        can_attack = False
        if can_attack:
            king_pos = find_king(board, color)
            if king_pos:
                state_data["bomb_anim"] = {"from_r": king_pos[0], "from_c": king_pos[1], "to_r": r, "to_c": c, "progress": 0.0}
            for fr in range(r_start, r_start+3):
                for fc in range(c_start, c_start+3):
                    if 0 <= fr < 8 and 0 <= fc < 8 and board[fr][fc]:
                        board[fr][fc] = None
        
    elif spell_type == "fortune":
        r, c = target1
        if board[r][c] and board[r][c]["type"][1] != "K":
            if random.random() < 0.5:
                new_piece = board[r][c].copy()
                free_spots = [(rr, cc) for rr in range(8) for cc in range(8) if board[rr][cc] is None]
                if free_spots:
                    rr, cc = random.choice(free_spots)
                    board[rr][cc] = new_piece
            else:
                board[r][c] = None
                
    elif spell_type == "nothing":
        pass
        
    elif spell_type == "trip":
        state_data["trip_mode"] = not state_data.get("trip_mode", False)
        
    elif spell_type == "betrayal":
        for r in range(8):
            for c in range(8):
                if board[r][c] and board[r][c]["type"] == color + "Q":
                    board[r][c] = None
        
    elif spell_type == "friendship":
        state_data["game_over"] = True
        state_data["end_message"] = "ДРУЖБА! Ничья."
        
    elif spell_type == "blindness":
        state_data["blind_turns"] = 2
        
    elif spell_type == "gold":
        r, c = target1
        if board[r][c]:
            board[r][c]["gold_projectile"] = True
            board[r][c]["armor"] = 2
        
    elif spell_type == "genghis":
        for r in range(8):
            for c in range(8):
                if board[r][c] and board[r][c]["type"][0] == color and board[r][c]["type"][1] != "K":
                    board[r][c]["type"] = color + "N"

    return True

                                    last_move = {"from_r": selected[0], "from_c": selected[1], "to_r": r, "to_c": c}
                                    selected = None; valid_moves = []
                                    continue
                                
                                selected = None; valid_moves = []
                                last_move = {"from_r": selected[0] if selected else -1, "from_c": -1, "to_r": r, "to_c": c}
                                turn = "b" if turn == "w" else "w"
                                move_counter += 1
                                new_check = is_king_in_check(board, turn, buffs)
                                if new_check and not check_status: check_alpha = 255
                                check_status = new_check
                                
                                status = is_checkmate_or_stalemate(board, turn, last_move, buffs)
                                if status == "checkmate":
                                    game_over = True; state = STATE_GAME_OVER; end_message = "МАТ!"
                                elif status == "stalemate":
                                    game_over = True; state = STATE_GAME_OVER; end_message = "ПАТ!"

                elif state == STATE_PROMOTION:
                    for rect, opt in promotion_rects:
                        if rect.collidepoint(mx, my):
                            r, c = promotion_pos
                            board[r][c]["type"] = turn + opt
                            state = STATE_PLAYING
                            turn = "b" if turn == "w" else "w"
                            move_counter += 1
                            new_check = is_king_in_check(board, turn, buffs)
                            if new_check and not check_status: check_alpha = 255
                            check_status = new_check
                            status = is_checkmate_or_stalemate(board, turn, last_move, buffs)
                            if status == "checkmate":
                                game_over = True; state = STATE_GAME_OVER; end_message = "МАТ!"
                            elif status == "stalemate":
                                game_over = True; state = STATE_GAME_OVER; end_message = "ПАТ!"
                            break

        if state == STATE_PLAYING and not is_book_open and not targeting_spell:
            current_time = time.time()
            
            if blind_turns > 0: blind_turns -= 1
                
            for fz in frozen_zones[:]:
                fz["turns"] -= 1
                if fz["turns"] <= 0:
                    for fr in range(fz["r"], fz["r"]+3):
                        for fc in range(fz["c"], fz["c"]+3):
                            if 0 <= fr < 8 and 0 <= fc < 8 and board[fr][fc]:
                                board[fr][fc]["frozen"] = 0
                    frozen_zones.remove(fz)

            if current_time - last_armor_spawn > ARMOR_SPAWN_TIME:
                last_armor_spawn = current_time
                rr = random.choice([3, 4]); cc = random.randint(0, 7)
                if board[rr][cc] is None and not any(i["r"] == rr and i["c"] == cc for i in items):
                    items.append({"r": rr, "c": cc, "type": "armor"})
            
            if current_time - last_bomb_spawn > BOMB_SPAWN_TIME:
                last_bomb_spawn = current_time
                rr = random.choice([3, 4]); cc = random.randint(0, 7)
                if board[rr][cc] is None and not any(i["r"] == rr and i["c"] == cc for i in items):
                    items.append({"r": rr, "c": cc, "type": "bomb"})

            for item in items[:]:
                for r in range(8):
                    for c in range(8):
                        if board[r][c] and r == item["r"] and c == item["c"]:
                            if item["type"] == "armor": board[r][c]["armor"] = 2
                            elif item["type"] == "bomb": board[r][c]["projectile"] = True
                            items.remove(item)
                            break

            if check_alpha > 0:
                check_alpha -= 5
                if check_alpha < 0: check_alpha = 0
                
            if proj_anim is not None:
                proj_anim["progress"] += 0.05
                if proj_anim["progress"] >= 1.0:
                    tr, tc = proj_anim["to_r"], proj_anim["to_c"]
                    target = board[tr][tc]
                    if target is not None:
                        if proj_anim["color"] == (255, 215, 0):
                            if target["type"][1] != "K": target["armor"] = 2
                        else:
                            if target["type"][1] != "K": board[tr][tc] = None
                    
                    turn = "b" if turn == "w" else "w"
                    move_counter += 1
                    new_check = is_king_in_check(board, turn, buffs)
                    if new_check and not check_status: check_alpha = 255
                    check_status = new_check
                    
                    status = is_checkmate_or_stalemate(board, turn, last_move, buffs)
                    if status == "checkmate":
                        game_over = True; state = STATE_GAME_OVER; end_message = "МАТ!"
                    elif status == "stalemate":
                        game_over = True; state = STATE_GAME_OVER; end_message = "ПАТ!"
                    proj_anim = None
                    
            if bomb_anim is not None:
                bomb_anim["progress"] += 0.05
                if bomb_anim["progress"] >= 1.0:
                    bomb_anim = None

        if state == STATE_MENU:
            draw_menu(screen, font_title, font_btn)
        elif state in [STATE_PLAYING, STATE_GAME_OVER, STATE_PROMOTION]:
            state_data = {
                "board": board, "selected": selected, "valid_moves": valid_moves, "turn": turn,
                "check_status": check_status, "check_alpha": check_alpha, "items": items,
                "is_book_open": is_book_open, "last_armor_spawn": last_armor_spawn, "last_bomb_spawn": last_bomb_spawn,
                "proj_anim": proj_anim, "bomb_anim": bomb_anim, "buffs": buffs, "blind_turns": blind_turns,
                "frozen_zones": frozen_zones, "teleports": teleports, "trip_mode": trip_mode,
                "available_spells": available_spells, "used_spells": used_spells, "targeting_spell": targeting_spell,
                "move_counter": move_counter, "last_spell_info": last_spell_info
            }
            spell_rects = draw_game(screen, state_data, font_title, font_info, font_coords, font_cjk_small)
            
            if state == STATE_GAME_OVER:
                draw_game_over(screen, font_title, font_btn, end_message)
            elif state == STATE_PROMOTION:
                promotion_rects = draw_promotion_menu(screen, font_title, font_btn, turn)

        pygame.display.flip()
        clock.tick(60)

    pygame.quit()
    sys.exit()

if __name__ == "__main__":
